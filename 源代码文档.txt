function loadSiZhenPatient() {
  const p = PATIENTS[(sizhenLevel - 1) % PATIENTS.length];
  document.getElementById('patient-symptoms').textContent = p.symptoms;

  // Parse gender and age from symptoms
  const match = p.symptoms.match(/患者：(.)，(\d+)岁/);
  if (match) {
    const genderLabel = match[1] === '男' ? '男患者' : '女患者';
    document.getElementById('patient-name').textContent = match[2] + '岁' + genderLabel;
    document.getElementById('patient-avatar').textContent = match[1] === '男' ? '👨' : '👩';
  }
  // Parse first symptom phrase for description
  const descMatch = p.symptoms.match(/\d+岁。(.+?)[，。]/);
  if (descMatch) {
    document.getElementById('patient-desc').textContent = descMatch[1];
  }

  const optsDiv = document.getElementById('sizhen-options');
  optsDiv.innerHTML = '';
  p.options.forEach((opt, idx) => {
    const div = document.createElement('div');
    div.className = 'sizhen-option';
    div.dataset.correct = opt.correct ? 'true' : 'false';
    div.onclick = function() { selectSiZhenOption(this, opt.correct, opt); };
    div.innerHTML = `
      <span class="opt-num">${String.fromCharCode(65 + idx)}</span>
      <div class="opt-info">
        <div class="opt-name">${opt.name}</div>
      </div>
    `;
    optsDiv.appendChild(div);
  });
}

    sizhenCorrect++;
    AudioSys.playCorrect();

function nextSizhen() {
  if (sizhenLevel >= 100) {
    showSizhenFinal();
    return;

  sizhenLevel++;
  document.getElementById('sizhen-level').textContent = sizhenLevel;
  showScreen('screen-sizhen');
  loadSiZhenPatient();
}

function startHerbGame() {
  const fid = gameState.currentFormula;
  const f = FORMULAS[fid];
  gameState.selectedHerbs = new Set();

  document.getElementById('game-formula-name').textContent = f.name;
  updateCollectionDisplay();

  // Build herb grid: correct herbs + distractors = 9 total
  const correctHerbs = [...f.herbs];
  const distractors = ALL_HERBS.filter(h => !correctHerbs.includes(h));
  // Shuffle distractors and pick enough to make 9
  const shuffledDistractors = shuffleArray([...distractors]);
  const numDistractors = Math.max(0, 9 - correctHerbs.length);
  const selectedDistractors = shuffledDistractors.slice(0, numDistractors);

  const gridHerbs = shuffleArray([...correctHerbs, ...selectedDistractors]);

  const grid = document.getElementById('herb-grid');
  grid.innerHTML = '';
  gridHerbs.forEach(herbName => {
    const isCorrect = correctHerbs.includes(herbName);
    const div = document.createElement('div');
    div.className = 'herb-item';
    div.dataset.name = herbName;
    div.dataset.correct = isCorrect;
    div.innerHTML = `
      <span class="herb-emoji">${getHerbEmoji(herbName)}</span>
      <span class="herb-name">${herbName}</span>
    `;
    div.onclick = () => selectHerb(div, herbName, isCorrect);
    grid.appendChild(div);
  });

  updateProgress();
  document.getElementById('btn-brew').disabled = true;
  showScreen('screen-game');

============================================================
杏林街市中医名方寻踪游戏 V1.0    第054页

}

function selectHerb(el, herbName, isCorrect) {
  if (el.classList.contains('correct') || el.classList.contains('wrong')) return;

  if (isCorrect) {

}

}

function shuffleArray(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// Parse composition string like "桂枝9g, 白芍9g, 大枣4枚" into [{herb, amount}]
function parseComposition(composition) {
  const result = [];
  // Split by either Chinese comma or English comma
  const parts = composition.split(/[,，]/);
  parts.forEach(part => {
    part = part.trim();
    if (!part) return;
    // Match herb name + amount: herb name is Chinese chars, amount is number+g/unit
    const m = part.match(/^([\u4e00-\u9fa5]+(?:\d+片|\d+枚)?)\s*([\d\.]+\s*[g克枚片ml]?.*)$/);
    if (m) {
      result.push({ herb: m[1].trim(), amount: m[2].trim() });
    } else {
      // Fallback: try to find first number as split point
      const numIdx = part.search(/[\d\.]/);
      if (numIdx > 0) {
        result.push({
          herb: part.substring(0, numIdx).trim(),
          amount: part.substring(numIdx).trim()
        });

============================================================
杏林街市中医名方寻踪游戏 V1.0    第056页
============================================================
      } else {
        result.push({ herb: part, amount: '' });
      }
    }
  });
  return result;
}

function wrapText(ctx, text, x, y, maxWidth, lineHeight) {
  const chars = text.split('');
  let line = '';
  let currentY = y;

  for (let i = 0; i < chars.length; i++) {
    const testLine = line + chars[i];
    const metrics = ctx.measureText(testLine);
    if (metrics.width > maxWidth && line.length > 0) {
      ctx.fillText(line, x, currentY);
      line = chars[i];
      currentY += lineHeight;
    } else {
      line = testLine;
    }
  }
  if (line) {
    ctx.fillText(line, x, currentY);
  }
  return currentY;
}

function copyLink() {
  const url = window.location.href;
  if (navigator.clipboard) {
    navigator.clipboard.writeText(url).then(() => {
      showToast('链接已复制，快去微信粘贴分享吧！');
    }).catch(() => {
      fallbackCopy(url);
    });
  } else {
    fallbackCopy(url);
  }
}

function fallbackCopy(text) {
  const ta = document.createElement('textarea');
  ta.value = text;
  ta.style.position = 'fixed';
  ta.style.opacity = '0';
  document.body.appendChild(ta);
  ta.select();
  try {
    document.execCommand('copy');
    showToast('链接已复制，快去微信粘贴分享吧！');
  } catch (e) {
    showToast('复制失败，请手动复制地址栏链接');
  }
  document.body.removeChild(ta);
}
